     1	name: Build Windows executable
     2	
     3	on:
     4	  push:
     5	    branches: [ main ]
     6	  workflow_dispatch:
     7	
     8	defaults:
     9	  run:
    10	    shell: bash
    11	
    12	jobs:
    13	  build-windows:
    14	    runs-on: windows-latest
    15	
    16	    steps:
    17	      - name: Checkout code
    18	        uses: actions/checkout@v4
    19	
    20	      - name: Set up Python
    21	        uses: actions/setup-python@v5
    22	        with:
    23	          python-version: '3.11'
    24	
    25	      - name: Install dependencies
    26	        run: |
    27	          set -e
    28	          python -m pip install --upgrade pip
    29	          pip install -r requirements-windows.txt
    30	          pip install pyinstaller
    31	
    32	      - name: Diagnose Open Babel install location
    33	        shell: pwsh
    34	        run: |
    35	          Write-Host "--- Recherche de obabel.exe sur le runner ---"
    36	          $obabelPath = (Get-Command obabel -ErrorAction SilentlyContinue).Source
    37	          Write-Host "obabel trouve via PATH : $obabelPath"
    38	
    39	          Write-Host "--- Recherche large dans site-packages ---"
    40	          $pyLib = python -c "import sysconfig; print(sysconfig.get_paths()['purelib'])"
    41	          Write-Host "site-packages : $pyLib"
    42	          Get-ChildItem -Path $pyLib -Recurse -Include "obabel*.exe","*.exe" -ErrorAction SilentlyContinue |
    43	            Where-Object { $_.FullName -match "openbabel" } |
    44	            ForEach-Object { Write-Host "EXE trouve : $($_.FullName)" }
    45	
    46	          Write-Host "--- Recherche des dossiers de donnees (BABEL_DATADIR probable) ---"
    47	          Get-ChildItem -Path $pyLib -Recurse -Directory -ErrorAction SilentlyContinue |
    48	            Where-Object { $_.Name -match "openbabel|BABEL" } |
    49	            ForEach-Object { Write-Host "Dossier trouve : $($_.FullName)" }
    50	
    51	      - name: Download AutoDock Vina (Windows binary)
    52	        run: |
    53	          set -e
    54	          mkdir -p docking/bin
    55	          curl -L -f -o docking/bin/vina.exe https://github.com/ccsb-scripps/AutoDock-Vina/releases/download/v1.2.5/vina_1.2.5_win.exe
    56	
    57	      - name: Verify Vina binary runs on this runner
    58	        shell: pwsh
    59	        run: |
    60	          $size = (Get-Item docking/bin/vina.exe).Length
    61	          Write-Host "Taille du fichier vina.exe : $size octets"
    62	          if ($size -lt 100000) {
    63	            Write-Error "Fichier vina.exe suspicieusement petit -- probablement un telechargement rate (page d'erreur au lieu du binaire)."
    64	            exit 1
    65	          }
    66	          & .\docking\bin\vina.exe --help
    67	          if ($LASTEXITCODE -ne 0) {
    68	            Write-Error "vina.exe ne s'execute pas correctement sur ce runner Windows x64 -- binaire incompatible ou corrompu."
    69	            exit 1
    70	          }
    71	
    72	      - name: Download Visual C++ Redistributable
    73	        run: |
    74	          set -e
    75	          mkdir -p vcredist
    76	          curl -L -f -o vcredist/vc_redist.x64.exe https://aka.ms/vs/17/release/vc_redist.x64.exe
    77	
    78	      - name: Diagnose Open Babel plugin/inchi location
    79	        shell: pwsh
    80	        run: |
    81	          $pyLib = python -c "import sysconfig; print(sysconfig.get_paths()['purelib'])"
    82	          Write-Host "--- Arborescence complete de openbabel (tous fichiers) ---"
    83	          Get-ChildItem -Path "$pyLib\openbabel" -Recurse -File |
    84	            ForEach-Object { Write-Host $_.FullName }
    85	
    86	          Write-Host "--- Recherche specifique de fichiers lies a inchi ---"
    87	          Get-ChildItem -Path "$pyLib\openbabel" -Recurse -File |
    88	            Where-Object { $_.Name -match "inchi" } |
    89	            ForEach-Object { Write-Host "INCHI trouve : $($_.FullName)" }
    90	
    91	          Write-Host "--- Test direct : obabel -L formats liste-t-il inchikey ? ---"
    92	          & "$pyLib\openbabel\bin\obabel.exe" -L formats | Select-String -Pattern "inchi"
    93	
    94	      - name: Build .exe with PyInstaller
    95	        run: |
    96	          set -e
    97	          SITE_PACKAGES=$(python -c "import sysconfig; print(sysconfig.get_paths()['purelib'].replace(chr(92), '/'))")
    98	          echo "site-packages resolu: $SITE_PACKAGES"
    99	          pyinstaller --noconfirm --onefile --windowed \
   100	            --name VinaStudio \
   101	            --paths src \
   102	            --add-data "src/reference_data;reference_data" \
   103	            --add-data "src/i18n;i18n" \
   104	            --add-data "docking/receptor;docking/receptor" \
   105	            --add-data "src/assets/fonts;assets/fonts" \
   106	            --add-data "image_theme;image_theme" \
   107	            --add-data "receptor_profiles;receptor_profiles" \
   108	            --add-binary "docking/bin/vina.exe;docking/bin" \
   109	            --add-data "$SITE_PACKAGES/openbabel/bin;obabel_runtime/bin" \
   110	            --add-data "$SITE_PACKAGES/openbabel.libs;obabel_runtime/bin" \
   111	            --add-data "$SITE_PACKAGES/openbabel_wheel.libs;obabel_runtime/bin" \
   112	            --add-data "$SITE_PACKAGES/openbabel/share/openbabel;obabel_runtime/share/openbabel" \
   113	            --add-data "vcredist/vc_redist.x64.exe;vcredist" \
   114	            --collect-all plip \
   115	            --collect-all openbabel \
   116	            run_gui.py
   117	
   118	      - name: Copy credit_du_logiciel next to the exe (required by _credits_root())

   120	        run: |
   121	          Copy-Item -Path "credit_du_logiciel" -Destination "dist\credit_du_logiciel" -Recurse -Force
   122	          if (-not (Test-Path "dist\credit_du_logiciel")) {
   123	            Write-Error "credit_du_logiciel absent apres copie -- abandon."
   124	            exit 1
   125	          }
   126	          Write-Host "OK : credit_du_logiciel copie dans dist\credit_du_logiciel"

   128	      - name: Build installer with Inno Setup
   129	        shell: pwsh
   130	        run: |
   131	          & "C:\Program Files (x86)\Inno Setup 6\ISCC.exe" "VinaStudio.iss"
   132
