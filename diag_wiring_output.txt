   203	    if not csv_path.exists():

   215	            (lower_fields[c] for c in _MOLECULE_COLUMN_CANDIDATES if c in lower_fields),

   224	            return None

   227	        for row in reader:

   229	                scores[str(row[molecule_col]).strip()] = float(row[score_col])

   407	def _extract_best_pose(ligand_pdbqt: Path, output_path: Path) -> Path:
   408	    best_pose_number = None
   409	    best_score = None
   410	    current_pose_number = None
   411	
   412	    with ligand_pdbqt.open("r", encoding="utf-8", errors="ignore") as handle:
   413	        for line in handle:
   414	            stripped = line.strip()
   415	
   416	            model_match = _MODEL_RE.match(stripped)
   417	            if model_match:
   418	                current_pose_number = int(model_match.group(1))
   419	                continue
   420	
   421	            score_match = _VINA_SCORE_RE.search(stripped)
   422	            if score_match and current_pose_number is not None:
   423	                current_score = float(score_match.group(1))
   424	                if best_score is None or current_score < best_score:
   425	                    best_score = current_score
   426	                    best_pose_number = current_pose_number
   427	
   428	    if best_pose_number is None:
   429	        best_pose_number = 1
   430	
   431	    inside = False
   432	    lines = []
   433	
   434	    model_pattern = re.compile(rf"^MODEL\s+{best_pose_number}\s*$", re.IGNORECASE)
   435	    next_model_pattern = re.compile(r"^MODEL\s+\d+", re.IGNORECASE)
   436	
   437	    with ligand_pdbqt.open("r", encoding="utf-8", errors="ignore") as handle:
   438	        for line in handle:
   439	            stripped = line.strip()
   440	
   441	            if model_pattern.match(stripped):
   442	                inside = True
   443	                lines.append(line)
   444	                continue
   445	
   446	            if inside and next_model_pattern.match(stripped):
   447	                break
   448	
   449	            if inside:
   450	                lines.append(line)
   451	                if stripped.upper() == "ENDMDL":
   452	                    break
   453	
   454	    if not lines:
   455	        raise RuntimeError(f"Impossible d'extraire une pose depuis {ligand_pdbqt}")
   456	
   457	    output_path.parent.mkdir(parents=True, exist_ok=True)
   458	    output_path.write_text("".join(lines), encoding="utf-8")
   459	
   460	    return output_path

   486	            )
   487	
   488	            self.finished.emit(
   489	                {
   490	                    "xml": plip_result["xml"],
   491	                    "complex": plip_result["complex"],
   492	                }
   493	            )
   494	
   495	        except Exception as exc:
   496	            self.failed.emit(str(exc))
   497

   499	# ============================================================================
   500	# BATCH — PLIP + diagramme 2D pour TOUS les hits, sans sélection manuelle
   501	# ============================================================================
   502	
   503	class BatchWorker(QObject):
   504	
   505	    progress = Signal(str, int, int)
   506	    finished = Signal(dict)
   507	
   508	    def __init__(self, hits):
   509	        super().__init__()
   510	        self.hits = hits
   511	
   512	    def run(self):
   513	        results = {}
   514	        errors = []
   515	        total = len(self.hits)
   516	
   517	        for index, hit in enumerate(self.hits, start=1):
   518	            self.progress.emit(f"{hit.molecule} ({hit.target})", index, total)
   519	
   520	            try:
   521	                pose_file = _selected_poses_dir(hit.target) / f"{hit.molecule}_pose_best.pdbqt"
   522	                _extract_best_pose(hit.ligand_pdbqt, pose_file)
   523	
   524	                runner = PLIPRunner(workdir=str(_plip_workdir(hit.target)))
   525	                runner.run(receptor=hit.receptor, pose=pose_file, ligand_id=hit.molecule)
   526	
   527	                interactions = compute_interactions_for_hit(hit)
   528	
   529	                output_png = _diagrams_dir(hit.target) / hit.molecule / f"{hit.molecule}_interaction_2d.png"
   530	                output_png.parent.mkdir(parents=True, exist_ok=True)
   531	
   532	                render_2d_diagram(
   533	                    _plip_report_xml(hit.target, hit.molecule),
   534	                    _plip_complex_pdb(hit.target, hit.molecule),
   535	                    output_png,
   536	                    hit.molecule,
   537	                )
   538	
   539	                key = hit_result_key(hit.target, hit.molecule)
   540	                results[key] = {
   541	                    "interactions": interactions,
   542	                    "diagram": str(output_png),
   543	                    "complex": str(_plip_complex_pdb(hit.target, hit.molecule)),
   544	                    "target": hit.target,
   545	                    "molecule": hit.molecule,
   546	                }
   547	
   548	            except Exception as exc:
   549	                errors.append(f"{hit.molecule} : {exc}")
   550	
   551	        self.finished.emit({"results": results, "errors": errors})
   552

   554	# ============================================================================
   555	# DIAGRAMME 2D — onglet "Interaction 2D"
   556	# (vrai diagramme moléculaire RDKit, via plip_2d_diagram.py)
   557	# ============================================================================
   558	
   559	def generate_2d_for_molecule(molecule: str, target: str = DEFAULT_TARGET) -> dict:
   560	    xml_file = _plip_report_xml(target, molecule)
   561	    complex_file = _plip_complex_pdb(target, molecule)
   562	
   563	    if not xml_file.exists() or not complex_file.exists():
   564	        raise FileNotFoundError(
   565	            f"Résultats PLIP introuvables pour {molecule} ({target}). "
   566	            f"Lance d'abord le calcul PLIP pour cette molécule."
   567	        )
   568	
   569	    output_png = _diagrams_dir(target) / molecule / f"{molecule}_interaction_2d.png"
   570	    output_png.parent.mkdir(parents=True, exist_ok=True)
   571	
   572	    render_2d_diagram(xml_file, complex_file, output_png, molecule)
   573	
   574	    return {"output": str(output_png)}

   577	# ============================================================================
   578	# EXPORT — bouton "Exporter tout"
   579	# ============================================================================
   580	
   581	def export_visualization_results(batch_payload: dict, destination_dir: Path) -> Path:
   582	    """Exporte TOUTES les cibles présentes dans batch_payload (MexB, MexR,
   583	    ou tout autre couple), pas une seule : avant, l'appelant ne passait
   584	    qu'une cible et le dossier d'une autre cible dockée dans la même
   585	    campagne n'était jamais copié."""
   586	    destination_dir = Path(destination_dir)
   587	    destination_dir.mkdir(parents=True, exist_ok=True)
   588	
   589	    results = batch_payload.get("results", {})
   590	
   591	    targets = sorted({
   592	        data.get("target", DEFAULT_TARGET) for data in results.values()
   593	    }) or [DEFAULT_TARGET]
   594	
   595	    # Une seule cible : structure plate, inchangée pour les campagnes
   596	    # mono-récepteur existantes. Plusieurs cibles : un sous-dossier par
   597	    # cible pour ne jamais mélanger leurs diagrammes/complexes.
   598	    single_target = len(targets) == 1
   599	
   600	    for target in targets:
   601	        target_dest = destination_dir if single_target else destination_dir / target
   602	
   603	        diagrams_src = _diagrams_dir(target)
   604	        if diagrams_src.exists():
   605	            shutil.copytree(diagrams_src, target_dest / "diagrammes_2d", dirs_exist_ok=True)
   606	
   607	        plip_src = _plip_workdir(target)
   608	        if plip_src.exists():
   609	            shutil.copytree(plip_src, target_dest / "plip_xml_complex", dirs_exist_ok=True)
   610	
   611	    summary_csv = destination_dir / "residus_interactions.csv"
   612	    with summary_csv.open("w", encoding="utf-8", newline="") as handle:
   613	        writer = csv.DictWriter(
   614	            handle,
   615	            fieldnames=["target", "molecule", "residue", "residue_id", "chain", "distance", "interaction_types"],
   616	        )
   617	        writer.writeheader()
   618	        for key, data in results.items():
   619	            molecule = data.get("molecule", key)
   620	            target = data.get("target", DEFAULT_TARGET)
   621	            for row in data["interactions"]["summary"]:
   622	                writer.writerow({"target": target, "molecule": molecule, **row})
   623	
   624	    return destination_dir
   625
